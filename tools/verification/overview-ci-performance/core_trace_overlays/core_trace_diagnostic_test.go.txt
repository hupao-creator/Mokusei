func TestCoreTraceSinkPreservesBoundedExactPrefix(t *testing.T) {
	writer := newBoundedCoreTrace(7)
	for _, part := range []string{"abc", "defghi", "jk"} {
		if n, err := writer.Write([]byte(part)); err != nil || n != len(part) {
			t.Fatalf("reader must keep draining: n=%d err=%v", n, err)
		}
	}
	if string(writer.bytes) != "abcdefg" || writer.seen != 11 || writer.dropped != 4 || cap(writer.bytes) != 7 {
		t.Fatalf("wrong retained prefix/accounting: %+v", writer)
	}
}

func TestCoreTraceDrainsAndDecodesProducerBoundaries(t *testing.T) {
	origin, err := observeClock()
	if err != nil {
		t.Fatal(err)
	}
	path := filepath.Join(t.TempDir(), "core.trace")
	diagnostic, err := startCoreTraceDiagnostic(path, origin)
	if err != nil {
		t.Fatal(err)
	}
	defer diagnostic.stop(nil)
	queue := newSourceQueue(origin, sourceQueueCapacity)
	queue.trace = diagnostic
	ctx, cancel := context.WithTimeout(context.Background(), time.Second)
	defer cancel()
	consumer := make(chan error, 1)
	go func() {
		consumer <- consumeSource(ctx, queue, func(task sourceTask) error {
			diagnostic.mark(tracePointStoreStart, task.Index, 0)
			endStore := diagnostic.region(traceSpanStore)
			endStore()
			diagnostic.mark(tracePointStoreReturn, task.Index, 0)
			return nil
		})
	}()
	start, err := rawUptimeNanoseconds()
	if err != nil {
		t.Fatal(err)
	}
	if err := produceSource(ctx, start+int64(time.Millisecond), 6, time.Millisecond, queue,
		func(int, sourceTask, error) {}); err != nil {
		t.Fatal(err)
	}
	if err := <-consumer; err != nil {
		t.Fatal(err)
	}
	if err := diagnostic.stop(errors.New("original measurement rejected its source deadline")); err != nil {
		t.Fatal(err)
	}
	var receipt struct {
		CaptureComplete bool                 `json:"captureComplete"`
		TraceBytes      int                  `json:"traceBytes"`
		RunError        string               `json:"runErrorBeforeTraceCleanup"`
		Overflow        bool                 `json:"overflow"`
		RawAnchors      []coreTraceRawAnchor `json:"rawAnchors"`
		PointCounts     map[string]uint64    `json:"pointCounts"`
	}
	data, err := os.ReadFile(path + ".json")
	if err != nil {
		t.Fatal(err)
	}
	if err := json.Unmarshal(data, &receipt); err != nil {
		t.Fatal(err)
	}
	if !receipt.CaptureComplete || receipt.Overflow || receipt.TraceBytes == 0 || receipt.RunError == "" {
		t.Fatalf("capture completion must be independent of measurement validity: %s", data)
	}
	if receipt.PointCounts[coreTracePointNames[tracePointWaitReturn]] != 6 || receipt.PointCounts[coreTracePointNames[tracePointEnqueued]] != 6 {
		t.Fatalf("source boundaries missing: %s", data)
	}
	for _, anchor := range receipt.RawAnchors {
		if anchor.RawBeforeNanoseconds <= 0 || anchor.RawAfterNanoseconds < anchor.RawBeforeNanoseconds || anchor.ClockError != "" {
			t.Fatalf("invalid raw-clock bracket: %+v", anchor)
		}
	}
	command := exec.Command(filepath.Join(runtime.GOROOT(), "bin", "go"), "tool", "trace", "-d=parsed", path)
	decoded, err := command.CombinedOutput()
	if err != nil {
		t.Fatalf("trace did not drain/decode: %v\n%s", err, decoded)
	}
	for _, marker := range []string{"overview.wait-return", "overview.enqueued", "overview.source.fifo-lock", "overview.worker.store-update"} {
		if !bytes.Contains(decoded, []byte(marker)) {
			t.Fatalf("decoded trace has no %s", marker)
		}
	}
}

func TestCoreTraceOverflowRetainsPrefixAndRejectsCompletion(t *testing.T) {
	origin, err := observeClock()
	if err != nil {
		t.Fatal(err)
	}
	path := filepath.Join(t.TempDir(), "core.trace")
	diagnostic, err := startCoreTraceWithLimit(path, origin, 128)
	if err != nil {
		t.Fatal(err)
	}
	err = diagnostic.stop(nil)
	if err == nil || !strings.Contains(err.Error(), "trace memory bound exceeded") {
		t.Fatalf("overflow must be a diagnostic failure: %v", err)
	}
	data, err := os.ReadFile(path)
	if err != nil {
		t.Fatal(err)
	}
	if len(data) != 128 {
		t.Fatalf("retained prefix length=%d", len(data))
	}
	var receipt struct {
		CaptureComplete, Overflow bool
		DroppedTraceBytes         int64
	}
	data, err = os.ReadFile(path + ".json")
	if err != nil {
		t.Fatal(err)
	}
	if err := json.Unmarshal(data, &receipt); err != nil {
		t.Fatal(err)
	}
	if receipt.CaptureComplete || !receipt.Overflow || receipt.DroppedTraceBytes <= 0 {
		t.Fatalf("overflow hidden from receipt: %s", data)
	}
}

func TestCoreTraceRequiresNewAbsoluteOutput(t *testing.T) {
	origin, err := observeClock()
	if err != nil {
		t.Fatal(err)
	}
	if _, err := startCoreTraceDiagnostic("relative.trace", origin); err == nil {
		t.Fatal("relative trace output was accepted")
	}
	path := filepath.Join(t.TempDir(), "core.trace")
	if err := os.WriteFile(path, []byte("existing evidence"), 0600); err != nil {
		t.Fatal(err)
	}
	if _, err := startCoreTraceDiagnostic(path, origin); err == nil {
		t.Fatal("existing evidence overwritten")
	}
	data, _ := os.ReadFile(path)
	if string(data) != "existing evidence" {
		t.Fatal("existing evidence changed")
	}
}

func TestCoreTraceEnvironmentIsExplicitAllowlist(t *testing.T) {
	t.Setenv("UNRELATED_TRACE_SECRET", "must-not-be-exported")
	values := readCoreTraceEnvironment()
	if len(values) != 4 {
		t.Fatalf("unexpected environment field count: %d", len(values))
	}
	for _, key := range []string{"GODEBUG", "GOGC", "GOMEMLIMIT", "GOMAXPROCS"} {
		want, present := os.LookupEnv(key)
		got, exists := values[key]
		if !exists || got.Set != present || got.Value != want {
			t.Errorf("incorrect whitelisted environment receipt for %s", key)
		}
	}
	if _, exists := values["UNRELATED_TRACE_SECRET"]; exists {
		t.Fatal("unrelated environment was exported")
	}
}
